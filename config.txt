# -----------------------------------------------------------------------------
# Feldolgozási PIPE parancssablonok
# -----------------------------------------------------------------------------
#
# A MAIN a négy feldolgozási módhoz külön shell parancssablont használ.
# Ha valamelyik kulcs hiányzik, üres vagy kommentelt, a MAIN a korábbi,
# programba épített alapértelmezett parancsot használja.
#
# Használható helyettesítők:
#
# {AUDIO_DEVICE}       shell-biztos ALSA/hangeszköz
# {AUDIO_SAMPLE_RATE}  mintavételi frekvencia
# {AUDIO_LF}           alsó vágási frekvencia
# {AUDIO_HF}           felső vágási frekvencia
# {AUDIO_ORDER}        szűrőrend
# {AUDIO_PEAK}         AGC célcsúcs
# {AUDIO_DELAY}        szűrőblokk/késleltetés
# {AUDIO_INVERT_ARG}   -i vagy üres
# {FILTER}             rs41_filter_stream.py útvonala
# {FILTER_COMMAND}     csak a szűrőprogram parancsa
# {FILTER_ARGS}        az aktuális szűrőargumentumok
# {RS41_MOD}           rs41_mod útvonala
# {MOD_COMMAND}        csak a modulátor/dekóder program parancsa
# {MOD_ARGS}           az aktuális rs41_mod argumentumok
# {DECODER}            RAW dekóder útvonala
# {PIPE_DELAY}         pipe_delay.pl útvonala
# {INPUT_FILE}         WAV/RAW/JSON bemeneti fájl
# {WAV_FILE}           élő WAV napló útvonala
# {WAV_LOG_PIPE}       WAV_LOG_ENABLED=1 esetén `| tee -- {WAV_FILE}`, különben üres
# {RLOG_FILE}          élő RAW napló
# {JLOG_FILE}          élő JSON napló
# {PERL}               aktuális Perl értelmező
# {PROJECT_DIR}        projektkönyvtár
#
# Élő felvétel dekódoló pipeline.
# Jelenlegi FILTER_COMMAND: ./rs41_filter_stream.py
# Jelenlegi FILTER_ARGS: -LF {AUDIO_LF} -HF {AUDIO_HF} -O {AUDIO_ORDER} -P {AUDIO_PEAK} -D {AUDIO_DELAY} -V
# Jelenlegi MOD_COMMAND: ./rs41_mod
# Jelenlegi MOD_ARGS: -vv -r {AUDIO_INVERT_ARG} /dev/stdin
# PIPE_RECORD=arecord -D {AUDIO_DEVICE} -t wav -f S16_LE -r {AUDIO_SAMPLE_RATE} -c 1 -q | {FILTER_COMMAND} {FILTER_ARGS} | {MOD_COMMAND} {MOD_ARGS} | tee -a {RLOG_FILE} | {DECODER} --json | tee -a {JLOG_FILE}
PIPE_RECORD=arecord -D {AUDIO_DEVICE} -t wav -f S16_LE -r {AUDIO_SAMPLE_RATE} -c 1 -q | {FILTER_COMMAND} {FILTER_ARGS} | ./rs41_mod.py | tee -a {RLOG_FILE} | {DECODER} --json | tee -a {JLOG_FILE}

# WAV-fájl feldolgozása.
# Jelenlegi FILTER_COMMAND: ./rs41_filter_stream.py
# Jelenlegi FILTER_ARGS: -LF {AUDIO_LF} -HF {AUDIO_HF} -O {AUDIO_ORDER} -P {AUDIO_PEAK} -D {AUDIO_DELAY} -V
# Jelenlegi MOD_COMMAND: ./rs41_mod
# Jelenlegi MOD_ARGS: -vv -r {AUDIO_INVERT_ARG} /dev/stdin
# PIPE_WAV=sox -- {INPUT_FILE} -t wav -b 16 -e signed-integer -c 1 -r {AUDIO_SAMPLE_RATE} - | {FILTER_COMMAND} {FILTER_ARGS} | {MOD_COMMAND} {MOD_ARGS} | {DECODER} --json
PIPE_WAV=sox -- {INPUT_FILE} -t wav -b 16 -e signed-integer -c 1 -r {AUDIO_SAMPLE_RATE} - | {FILTER_COMMAND} {FILTER_ARGS} | ./rs41_mod.py | {DECODER} --json

# Demodulátor-választás:
# - A jelenlegi alapértelmezés {MOD_COMMAND} {MOD_ARGS}, vagyis ./rs41_mod.
# - Ha a PIPE_RECORD/PIPE_WAV sor hiányzik vagy kommentelt, a MAIN szintén
#   a beépített ./rs41_mod + dinamikus MOD_ARGS parancsot használja.
# - A Python változathoz a megfelelő pipeline-rész legyen:
#     | ./rs41_mod.py |
#   Az rs41_mod.py nem igényli a {MOD_ARGS} argumentumokat, és az invertált
#   illetve nem invertált jelet automatikusan felismeri.


# RAW-fájl dekódolása.
# FILTER_COMMAND/FILTER_ARGS: ebben a módban nincs használva.
# MOD_COMMAND/MOD_ARGS: ebben a módban nincs használva.
PIPE_RAW=cat -- {INPUT_FILE} | {DECODER} --json | {PIPE_DELAY}

# JSON-fájl visszajátszása.
# FILTER_COMMAND/FILTER_ARGS: ebben a módban nincs használva.
# MOD_COMMAND/MOD_ARGS: ebben a módban nincs használva.
PIPE_JSON=cat -- {INPUT_FILE} | {PIPE_DELAY}

# Az élő felvétel WAV-naplója: 0=kikapcsolva, 1=bekapcsolva.
# Ha a bejegyzés hiányzik, üres vagy kommentelt, az alapérték 1.
#WAV_LOG_ENABLED=1

# Az élő vétel külön WAV-mentő és helyi monitorozó ága.
# A {WAV_LOG_PIPE} WAV_LOG_ENABLED=1 esetén a tee WAV-ág, különben üres.
# BSD/Hurd portnál ezt kell az adott hangrendszer parancsára cserélni.
RECORD_MONITOR_COMMAND=arecord -D {AUDIO_DEVICE} -t wav -f S16_LE -r {AUDIO_SAMPLE_RATE} -c 1 -q - {WAV_LOG_PIPE} | aplay -q

# -----------------------------------------------------------------------------
# BT és SondeHub worker indítás
# -----------------------------------------------------------------------------
#
# A MAIN ezeket általános POSIX fork/pipe/setsid/exec mechanizmussal,
# `sh -c` parancsként indítja.
#
# Használható helyettesítők:
#
# {PERL}
# {WORKER}
# {BT_WORKER}
# {SONDEHUB_WORKER}
# {PIPE_CONNECT}
# {PROJECT_DIR}
#
# Linuxon a Bluetooth worker PTY-ját a util-linux `script` biztosítja.
# BSD/Hurd rendszeren ez a sor az adott rendszer PTY-wrapperére cserélhető.
BT_WORKER_COMMAND=script -qefc "exec {PERL} {BT_WORKER} 2>&1" /dev/null

# A SondeHub worker nem igényel PTY-t.
SONDEHUB_WORKER_COMMAND=exec {PERL} {SONDEHUB_WORKER} 2>&1
